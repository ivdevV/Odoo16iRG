## Runner (docker compose service odoo_local has no built image: Debian 11 apt 404, so docker run on odoo:16.0, as prior missions)
#!/bin/sh
# usage: run_odoo.sh <db> <odoo flags...>
DB=$1; shift
docker run --rm --network odoo16irg_local_default --name odoo_test_oficialidad_types \
  -v "<repo>/addons-extra:/mnt/extra-addons:ro" \
  -v "<repo>/etc/odoo/odoo.local.conf:/etc/odoo/odoo.conf:ro" \
  odoo:16.0 odoo -c /etc/odoo/odoo.conf -d $DB "$@" \
  --stop-after-init --workers=0 --http-port=18069

## Deps bootstrap (fresh DB test_irg_oficialidad_types)
run_odoo.sh test_irg_oficialidad_types -i isep_gradebook --without-demo=all   # isep_control_escolar has undeclared dep on isep_gradebook
run_odoo.sh test_irg_oficialidad_types -i isep_control_escolar --without-demo=all

## RED command
run_odoo.sh test_irg_oficialidad_types -i irg_student_oficialidad_documents --test-enable --test-tags=/irg_student_oficialidad_documents --log-level=test
